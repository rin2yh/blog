{{- /* Resize a Hugo image Resource for display, with SVG bypass. Input dict: - resource (Resource, required) - op (string, "Fill" or "Resize"; default "Fill") - spec (string, e.g. "640x360 q85"; required for raster) Returns dict: { url, width, height } Hugo's image-processing methods (.Fill / .Resize / .Width / .Height) only support raster formats with a Go decoder: jpeg, png, gif, webp, tiff. SVG, BMP, ICO, AVIF, etc. fall through to the original URL with width/height = 0 — pass them through your own pipeline if you need processed variants of those formats. */ -}} {{- $r := .resource -}} {{- $url := $r.RelPermalink -}} {{- $w := 0 -}} {{- $h := 0 -}} {{- /* Hugo's .Width/.Height aren't exposed on SVG resources, so we have to gate even the intrinsic-dimension read on ResourceType+SubType. */ -}} {{- $processable := and (eq $r.ResourceType "image") (ne $r.MediaType.Type "image/svg+xml") -}} {{- if $processable -}} {{- $w = $r.Width -}} {{- $h = $r.Height -}} {{- end -}} {{- if and $processable .spec -}} {{- $op := .op | default "Fill" -}} {{- $processed := $r -}} {{- if eq $op "Resize" -}} {{- $processed = $r.Resize .spec -}} {{- else -}} {{- $processed = $r.Fill .spec -}} {{- end -}} {{- $url = $processed.RelPermalink -}} {{- $w = $processed.Width -}} {{- $h = $processed.Height -}} {{- end -}} {{- return dict "url" $url "width" $w "height" $h -}}
