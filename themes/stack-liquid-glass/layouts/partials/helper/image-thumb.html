{{- /* Resize a Hugo image Resource for display, with SVG bypass. Input dict: - resource (Resource, required) - op (string, "Fill" or "Resize"; default "Fill") - spec (string, e.g. "640x360 q85"; required for raster) Returns dict: { url, width, height } Hugo's image-processing methods (.Fill / .Resize / .Width / .Height) only run on raster formats with a Go decoder. Anything outside the whitelist below (SVG, BMP, ICO, etc.) falls through to the original URL with width/height = 0. */ -}} {{- $r := .resource -}} {{- $url := $r.RelPermalink -}} {{- $w := 0 -}} {{- $h := 0 -}} {{- $processable := and (eq $r.ResourceType "image") (in (slice "jpeg" "png" "gif" "webp" "tiff" "avif") $r.MediaType.SubType) -}} {{- if $processable -}} {{- $w = $r.Width -}} {{- $h = $r.Height -}} {{- end -}} {{- if and $processable .spec -}} {{- $op := .op | default "Fill" -}} {{- $processed := $r -}} {{- if eq $op "Resize" -}} {{- $processed = $r.Resize .spec -}} {{- else -}} {{- $processed = $r.Fill .spec -}} {{- end -}} {{- $url = $processed.RelPermalink -}} {{- $w = $processed.Width -}} {{- $h = $processed.Height -}} {{- end -}} {{- return dict "url" $url "width" $w "height" $h -}}
