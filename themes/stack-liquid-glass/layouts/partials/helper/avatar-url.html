{{- /* Resolve a local avatar resource to a URL. Input: dict - src (string, required) — path passed to resources.Get - fillSize (string, optional) — e.g. "192x192 q85"; omitted/empty = no resize SVG resources don't support .Fill, so they're returned as-is. Raster resources are processed when fillSize is set, untouched otherwise. Returns "" if the resource is missing. */ -}} {{- $url := "" -}} {{- with resources.Get .src -}} {{- if eq .MediaType.SubType "svg" -}} {{- $url = .RelPermalink -}} {{- else if $.fillSize -}} {{- $url = (.Fill $.fillSize).RelPermalink -}} {{- else -}} {{- $url = .RelPermalink -}} {{- end -}} {{- end -}} {{- return $url -}}
