{{- /* Resolve a local avatar resource to a URL. Input: dict - src (string, required) — path passed to resources.Get - fillSize (string, optional) — e.g. "192x192 q85"; omitted/empty = no resize Only raster image resources are processed via .Fill; SVGs and any non-image resource (misconfigured CSS / JSON / etc.) fall back to the original URL so a typo in config never crashes the build. Returns "" if the resource is missing. */ -}} {{- $url := "" -}} {{- with resources.Get .src -}} {{- if and (eq .ResourceType "image") (ne .MediaType.Type "image/svg+xml") $.fillSize -}} {{- $url = (.Fill $.fillSize).RelPermalink -}} {{- else -}} {{- $url = .RelPermalink -}} {{- end -}} {{- end -}} {{- return $url -}}
