{{- /* Resolve link target for a Page. Input : a Page (.) Output: dict { url, target, rel, icon, isExternal } */ -}} {{- $url := .RelPermalink -}} {{- $target := "" -}} {{- $rel := "" -}} {{- $icon := "" -}} {{- with .Params.externalUrl -}} {{- $url = . -}} {{- $target = "_blank" -}} {{- $rel = "noopener noreferrer" -}} {{- if strings.Contains . "zenn.dev" -}} {{- $icon = "zenn-logo" -}} {{- else if strings.Contains . "speakerdeck.com" -}} {{- $icon = "speakerdeck" -}} {{- else -}} {{- $icon = "external" -}} {{- end -}} {{- end -}} {{- return (dict "url" $url "target" $target "rel" $rel "icon" $icon "isExternal" (ne $target "")) -}}
